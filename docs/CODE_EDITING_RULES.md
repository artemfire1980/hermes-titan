# Правила работы с кодом

## 1. Создать файл заново
cat > ~/путь/к/файлу.py << 'PYEOF'
# полный код файла
PYEOF

## 2. Заменить кусок в существующем файле
python3 << 'PYEOF'
path = "/home/khadas/путь/к/файлу.py"
with open(path) as f:
    code = f.read()
old = '''старый код'''
new = '''новый код'''
code = code.replace(old, new)
with open(path, "w") as f:
    f.write(code)
print("✅ Готово")
PYEOF

## 3. Заменить строку
sed -i 's/старая_строка/новая_строка/g' ~/путь/к/файлу.py

## 4. Добавить строку в конец
echo 'новая строка' >> ~/путь/к/файлу.py

## 5. Удалить файл
rm -f ~/путь/к/файлу.py

## 6. Проверить синтаксис после правки
python3 -m py_compile ~/путь/к/файлу.py && echo "✅ OK" || echo "❌ Ошибка"

## 7. Если файл защищён (chmod 444)
chmod 600 ~/путь/к/файлу.py
# ... правки ...
chmod 444 ~/путь/к/файлу.py

## 8. Поиск по файлам
grep -n "что_ищем" ~/путь/к/файлу.py
grep -rn "что_ищем" ~/путь/к/директории/ --include="*.py"

## 9. Показать содержимое
cat ~/путь/к/файлу.py
head -20 ~/путь/к/файлу.py
tail -20 ~/путь/к/файлу.py
sed -n '100,120p' ~/путь/к/файлу.py
